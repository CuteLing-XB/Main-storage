        -- 建立广播文件夹用于存储“已启动脚本的用户列表”（确保脚本用户之间互通）
        local scriptUsersFolder = ReplicatedStorage:FindFirstChild("BagahScriptUsers")
        if not scriptUsersFolder then
            scriptUsersFolder = Instance.new("Folder")
            scriptUsersFolder.Name = "BagahScriptUsers"
            pcall(function()

        -- 在 ReplicatedStorage 注册当前玩家为“脚本使用者”
        local function registerSelfAsUser()
            if scriptUsersFolder and not scriptUsersFolder:FindFirstChild(LocalPlayer.Name) then
                local userValue = Instance.new("BoolValue")
                userValue.Name = LocalPlayer.Name
                userValue.Value = true
                pcall(function()

                end)

                LocalPlayer.Character:SetAttribute("BagahScriptUser", true)
            end

        local function ensureHeadTag(character, tagName, text, fontColor)
            if not character then return nil end
            local head = character:FindFirstChild("Head")
            if not head then return nil end

            local billboard = head:FindFirstChild(tagName)
            if not billboard then
                billboard = Instance.new("BillboardGui")
                billboard.Name = tagName
                billboard.Size = UDim2.new(0, 220, 0, 45)
                billboard.StudsOffset = Vector3.new(0, 3, 0) -- 头顶高度 3 单位
                billboard.AlwaysOnTop = true -- 穿透遮挡、隔墙透视可见
                billboard.LightInfluence = 0 -- 不受环境光影响

                local textLabel = Instance.new("TextLabel")
                textLabel.Name = "TagLabel"
                textLabel.Size = UDim2.new(1, 0, 1, 0)
                textLabel.BackgroundTransparency = 1
                textLabel.Text = text
                textLabel.TextScaled = true
                textLabel.Font = Enum.Font.GothamBold
                textLabel.TextStrokeTransparency = 0.1 -- 黑色厚描边
                textLabel.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
                textLabel.TextColor3 = fontColor or Color3.fromRGB(255, 255, 255)
                textLabel.Parent = billboard

                billboard.Parent = head
            end

            billboard.Adornee = head
            return billboard:FindFirstChild("TagLabel")
        end

        -- 删除标签
        local function removeHeadTag(character, tagName)
            if character and character:FindFirstChild("Head") then
                local tag = character.Head:FindFirstChild(tagName)
                if tag then tag:Destroy() end
            end
        end

        -- 判断某个玩家是否启动了本脚本
        local function isScriptUser(player)
            if not player then return false end
            
            if scriptUsersFolder and scriptUsersFolder:FindFirstChild(player.Name) then
                return true
            end
            
            if player.Character and player.Character:GetAttribute("BagahScriptUser") == true then
                return true
            end

            return false
        end

                -- 核心改动：如果是自己，清理标签并直接跳过，不在自己头顶渲染任何东西
                if player == LocalPlayer then
                    if player.Character then
                        removeHeadTag(player.Character, "DevTitleTag")
                        removeHeadTag(player.Character, "UserTitleTag")
                    end
                elseif player.Character then
                    -- 针对其他玩家进行检测渲染：
                    if player.Name == DEV_USERNAME then
                        -- 1. [开发者] 无论开没开脚本，其他脚本用户看开发者都是【彩虹开发者】
                        local textLabel = ensureHeadTag(player.Character, "DevTitleTag", "👑 汉化作者 👑")

                            local hue = (tick() * COLOR_SPEED) % 1
                            textLabel.TextColor3 = Color3.fromHSV(hue, 1, 1) -- 彩虹流光
                        end
                    elseif isScriptUser(player) then
                        -- 2. [其他脚本使用者] 显示【脚本用户】标签
                        ensureHeadTag(player.Character, "UserTitleTag", "⚡ 脚本用户 ⚡", Color3.fromRGB(0, 255, 127))
                    else
                        -- 3. [普通玩家] 没开脚本的路人：清除标签
                        removeHeadTag(player.Character, "UserTitleTag")
                    end
                end
            end
        end)
    end)

    local function translateText(text)
        if not text or type(text) ~= "string" then return text end
        if Translations[text] then return Translations[text] end
        for en, cn in pairs(Translations) do
            if text:find(en) then return text:gsub(en, cn) end
        end
        return text
    end

    local function setupTranslationEngine()
        local success, err = pcall(function()
            local oldIndex = getrawmetatable(game).__newindex
            setreadonly(getrawmetatable(game), false)
            getrawmetatable(game).__newindex = newcclosure(function(t, k, v)
                if (t:IsA("TextLabel") or t:IsA("TextButton") or t:IsA("TextBox")) and k == "Text" then
                    v = translateText(tostring(v))
                end
                return oldIndex(t, k, v)
            end)
            setreadonly(getrawmetatable(game), true)
        end)
        
        if not success then
            local translated = {}
            local function scanAndTranslate()
                for _, gui in ipairs(game:GetService("CoreGui"):GetDescendants()) do
                    if (gui:IsA("TextLabel") or gui:IsA("TextButton") or gui:IsA("TextBox")) and not translated[gui] then
                        pcall(function()
                            local text = gui.Text
                            if text and text ~= "" then
                                local translatedText = translateText(text)
                                if translatedText ~= text then gui.Text = translatedText; translated[gui] = true end
                            end
                        end)
                    end
                end
            end
            while true do scanAndTranslate(); task.wait(3) end
        end
    end
