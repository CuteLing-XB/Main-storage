-- ==================== Rayfield 加载 ====================
local function LoadRayfield()
    local urls = {
        "https://sirius.menu/rayfield",
        "https://raw.githubusercontent.com/SiriusSoftwareLtd/Rayfield/refs/heads/main/source.lua",
    }
    local lastError = "未知错误"
    for _, url in ipairs(urls) do
        local ok, result = pcall(function()
            local source = game:HttpGet(url)
            if type(source) ~= "string" or #source < 1000 then error("UI 库源码异常") end
            local chunk, err = loadstring(source)
            if not chunk then error(err or "编译失败") end
            return chunk()
        end)
        if ok and type(result) == "table" and type(result.CreateWindow) == "function" then return result end
        lastError = tostring(result)
        task.wait(0.15)
    end
    error("Rayfield 加载失败: " .. lastError)
end

local RemotePlayerEvent, RemotePlayerFunc
local function ResolveRemotes()
    local remoteFolder = ReplicatedStorage:FindFirstChild("Remote")
    if not remoteFolder then return false end
    RemotePlayerEvent = RemotePlayerEvent or remoteFolder:FindFirstChild("PlayerEvent")
    RemotePlayerFunc = RemotePlayerFunc or remoteFolder:FindFirstChild("PlayerFunc")
    return RemotePlayerEvent ~= nil and RemotePlayerFunc ~= nil
end
task.spawn(function() for _ = 1, 200 do if ResolveRemotes() then return end task.wait(0.1) end end)
local function SafeFirePlayerEvent(...)
    if not RemotePlayerEvent then ResolveRemotes() end
    if not RemotePlayerEvent then return false, "Remote.PlayerEvent 未找到" end
    return pcall(function(...) RemotePlayerEvent:FireServer(...) end, ...)
end

local function GetVehicle()
    local cached = VehicleCache.Seat
    if cached and cached.Parent then
        local char = LocalPlayer.Character
        if char and cached:IsDescendantOf(char) then return cached end
    end
    local char = LocalPlayer.Character
    if not char then return nil end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then
        local seat = hum.SeatPart
        if seat and (seat:IsA("VehicleSeat") or seat:IsA("Seat")) then
            VehicleCache.Seat = seat
            return seat
        end
    end
    for _, desc in ipairs(char:GetChildren()) do
        if desc:IsA("Weld") or desc:IsA("WeldConstraint") or desc:IsA("Motor6D") then
            local p1, p0 = desc.Part1, desc.Part0
            if p1 and (p1:IsA("VehicleSeat") or p1:IsA("Seat")) then VehicleCache.Seat = p1; return p1 end
            if p0 and (p0:IsA("VehicleSeat") or p0:IsA("Seat")) then VehicleCache.Seat = p0; return p0 end
        end
    end
    local now = os.clock()
    if now - (VehicleCache.LastDeepSeatScan or 0) >= 0.5 then
        VehicleCache.LastDeepSeatScan = now
        for _, desc in ipairs(char:GetDescendants()) do
            if desc:IsA("Weld") or desc:IsA("WeldConstraint") or desc:IsA("Motor6D") then
                local p1, p0 = desc.Part1, desc.Part0
                if p1 and (p1:IsA("VehicleSeat") or p1:IsA("Seat")) then VehicleCache.Seat = p1; return p1 end
                if p0 and (p0:IsA("VehicleSeat") or p0:IsA("Seat")) then VehicleCache.Seat = p0; return p0 end
            end
        end
    end
    VehicleCache.Seat = nil
    return nil
end

local function GetVehicleContext(forceRefresh)
    local now = os.clock()
    if not forceRefresh and (now - VehicleCache.LastUpdate) < 0.10 then
        local seat = VehicleCache.Seat
        if seat and seat.Parent then return VehicleCache.Seat, VehicleCache.Model, VehicleCache.Root end
    end
    local seat = GetVehicle()
    if not seat then InvalidateVehicleCache(); VehicleCache.LastUpdate = now; return nil, nil, nil end
    local model = seat:FindFirstAncestorWhichIsA("Model") or seat.Parent
    local root = seat.AssemblyRootPart or (model and model.PrimaryPart) or seat
    VehicleCache.Seat = seat; VehicleCache.Model = model; VehicleCache.Root = root; VehicleCache.LastUpdate = now
    return seat, model, root
end

local function FindVehicleConfigModule(seat)
    if not seat then return nil end
    local model = seat:FindFirstAncestorWhichIsA("Model") or seat.Parent
    if not model then return nil end
    for _, desc in ipairs(model:GetDescendants()) do
        if desc:IsA("ModuleScript") then
            local name = string.lower(desc.Name)
            if name == "config" or name:find("config", 1, true) or name:find("vehicle", 1, true) then return desc end
        end
    end
    return nil
end

local function GetControlRoot()
    local seat, model, root = GetVehicleContext()
    if root then return root, true end
    if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
        return LocalPlayer.Character.HumanoidRootPart, false
    end
    return nil, false
end

-- ===== 修复：传送函数现在强制使用角色根部件（如果载具根不存在） =====
local function SafeStepTeleport(targetPos)
    local root, isVeh = GetControlRoot()
    if not root then
        -- 如果 GetControlRoot 返回 nil，尝试直接获取角色 HumanoidRootPart
        local char = LocalPlayer.Character
        if char then
            root = char:FindFirstChild("HumanoidRootPart")
        end
        if not root then
            SendNotify("传送失败", "无法找到角色根部件，请重试。", 2)
            return
        end
    end
    local startPos = root.Position
    local dist = (targetPos - startPos).Magnitude
    if dist < 15 then
        root.CFrame = CFrame.new(targetPos)
        pcall(function() root.AssemblyLinearVelocity = Vector3.zero; root.AssemblyAngularVelocity = Vector3.zero end)
        return
    end
    local steps = math.clamp(math.floor(dist / 18), 2, 80)
    for i = 1, steps do
        local alpha = i / steps
        root.CFrame = CFrame.new(startPos:Lerp(targetPos, alpha))
        pcall(function() root.AssemblyLinearVelocity = Vector3.zero; root.AssemblyAngularVelocity = Vector3.zero end)
        task.wait(0.01)
    end
    root.CFrame = CFrame.new(targetPos)
    pcall(function() root.AssemblyLinearVelocity = Vector3.zero; root.AssemblyAngularVelocity = Vector3.zero end)
end

local function DoForwardTP()
    local dist = States.TeleportDistance
    local cam = Camera
    if not cam then return end
    local pos = cam.CFrame.Position + cam.CFrame.LookVector * dist
    local root, isVehicle = GetControlRoot()
    if root then
        local currentCF = root.CFrame
        local _, yaw, _ = currentCF:ToOrientation()
        local newCF = CFrame.new(pos) * CFrame.Angles(0, yaw, 0)
        root.CFrame = newCF
        if States.FreezeEnabled then States.FreezePosition = newCF end
        pcall(function() root.AssemblyLinearVelocity = Vector3.zero; root.AssemblyAngularVelocity = Vector3.zero end)
    else
        SendNotify("传送失败", "未找到角色根部件。", 2)
    end
end

local function MakeDraggable(guiObject)
    local dragging, dragInput, dragStart, startPos = false, nil, nil, nil
    guiObject.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = input.Position; startPos = guiObject.Position
            input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)
        end
    end)
    guiObject.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end
    end)
    UserInput.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            guiObject.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local function SpringLerp(current, target, speed, dt)
    return current + (target - current) * math.clamp(speed * dt, 0, 1)
end

local function IsPolicePlayer(player)
    if not player then return false end
    if player.TeamColor then
        local c = player.TeamColor.Color
        local h, s, v = Color3.toHSV(c)
        if s >= 0.25 and v >= 0.25 and h >= 0.47 and h <= 0.74 then return true end
        if c.B >= 0.50 and c.B > c.R * 1.12 and c.B > c.G * 1.01 then return true end
    end
    local teamName = player.Team and string.lower(player.Team.Name or "") or ""
    local roleName = string.lower(tostring(player:GetAttribute("Role") or player:GetAttribute("Team") or ""))
    local value = teamName .. " " .. roleName
    return value:find("police", 1, true) ~= nil or value:find("cop", 1, true) ~= nil
        or value:find("sheriff", 1, true) ~= nil or value:find("law", 1, true) ~= nil
        or value:find("警察", 1, true) ~= nil or value:find("警员", 1, true) ~= nil or value:find("执法", 1, true) ~= nil
end

local function GetLocalDetectionPosition()
    local char = LocalPlayer.Character
    if not char then return nil end
    local root = char:FindFirstChild("HumanoidRootPart")
    if root and root.Parent then return root.Position end
    return nil
end

local function GetNearestPlayer()
    local myPos = GetLocalDetectionPosition()
    if not myPos then return nil, nil, false end
    local maxDist = math.max(1, tonumber(States.WarningDistance) or 50)
    local nearest, bestDist, armed = nil, maxDist + 0.001, false
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local char = p.Character
            local root = char:FindFirstChild("HumanoidRootPart")
            local hum = char:FindFirstChildOfClass("Humanoid")
            if root and hum and hum.Health > 0 then
                local d = (root.Position - myPos).Magnitude
                if d <= maxDist and d < bestDist then
                    nearest = p; bestDist = d; armed = char:FindFirstChildOfClass("Tool") ~= nil
                end
            end
        end
    end
    return nearest, bestDist, armed
end

local function PlayWarningBeep(distance)
    if not States.WarningSound then return end
    local now = os.clock()
    if now < RadarAnim.beepCooldown then return end
    local maxDist = math.max(1, tonumber(States.WarningDistance) or 50)
    local closeness = math.clamp(1 - distance / maxDist, 0, 1)
    RadarAnim.beepCooldown = now + BEEP_MIN_INTERVAL + (1 - closeness) * 0.6
    States.WarningSound:Play()
end

local function UpdateWarningData(player, distance, isArmed)
    if not player then return end
    States.WarningCurrentPlayer = player
    States.WarningCurrentTargetId = player.UserId
    States.WarningLastDistance = distance
end

function StopWarningIsland()
    States.WarningCurrentPlayer = nil; States.WarningCurrentTargetId = nil; States.WarningLastDistance = math.huge
    if States.WarningSound then pcall(function() States.WarningSound:Stop() end) end
end

function HideWarningIslandCompletely()
    local token = RadarAnim.closeToken + 1
    RadarAnim.closeToken = token
    States.WarningCurrentPlayer = nil; States.WarningCurrentTargetId = nil; States.WarningLastDistance = math.huge
    if States.WarningSound then pcall(function() States.WarningSound:Stop() end) end
    if States.WarningCard and States.WarningCard.Parent then
        RadarAnim.targetY = RADAR_HIDE_Y
        task.delay(0.3, function()
            if RadarAnim.closeToken == token and not States.ProximityWarning then DestroyWarningUI() end
        end)
    end
    States.WarningVisible = false
end

RunService.RenderStepped:Connect(function(dt)
    if not States.ProximityWarning or not States.WarningCard or not States.WarningCard.Parent then return end
    local card = States.WarningCard
    local hasTarget = States.WarningCurrentPlayer ~= nil and States.WarningCurrentTargetId ~= nil

    if hasTarget then
        RadarAnim.targetWidth = RADAR_CARD_WIDTH_EXPANDED; RadarAnim.targetHeight = RADAR_CARD_HEIGHT_EXPANDED
        RadarAnim.targetCorner = RADAR_CARD_CORNER_EXPANDED; RadarAnim.targetY = RADAR_SHOW_Y
        States.WarningExpanded = true; States.WarningIslandState = "Expanded"
    else
        RadarAnim.targetWidth = RADAR_CARD_WIDTH_IDLE; RadarAnim.targetHeight = RADAR_CARD_HEIGHT_IDLE
        RadarAnim.targetCorner = RADAR_CARD_CORNER_IDLE; RadarAnim.targetY = RADAR_SHOW_Y
        States.WarningExpanded = false; States.WarningIslandState = "Idle"
    end

    local lf = math.clamp(ANIM_LERP_SPEED * dt, 0, 1)
    RadarAnim.currentWidth = SpringLerp(RadarAnim.currentWidth, RadarAnim.targetWidth, ANIM_LERP_SPEED, dt)
    RadarAnim.currentHeight = SpringLerp(RadarAnim.currentHeight, RadarAnim.targetHeight, ANIM_LERP_SPEED, dt)
    RadarAnim.currentCorner = SpringLerp(RadarAnim.currentCorner, RadarAnim.targetCorner, ANIM_LERP_SPEED, dt)
    RadarAnim.currentY = SpringLerp(RadarAnim.currentY, RadarAnim.targetY, ANIM_LERP_SPEED, dt)

    card.Size = UDim2.new(0, math.round(RadarAnim.currentWidth), 0, math.round(RadarAnim.currentHeight))
    card.Position = UDim2.new(0.5, 0, 0, math.round(RadarAnim.currentY))
    local corner = card:FindFirstChild("Corner")
    if corner then corner.CornerRadius = UDim.new(0, math.round(RadarAnim.currentCorner)) end

    local expandThreshold = (RADAR_CARD_HEIGHT_IDLE + RADAR_CARD_HEIGHT_EXPANDED) / 2
    SetWarningTextVisible(RadarAnim.currentHeight > expandThreshold)

    if hasTarget and States.WarningTitle and States.WarningDistanceText and States.WarningStatusText then
        local police = IsPolicePlayer(States.WarningCurrentPlayer)
        local displayName = States.WarningCurrentPlayer.DisplayName or States.WarningCurrentPlayer.Name
        States.WarningTitle.Text = police and ("警察靠近  ·  " .. displayName) or ("玩家靠近  ·  " .. displayName)
        States.WarningTitle.TextColor3 = police and Color3.fromRGB(224, 238, 255) or Color3.fromRGB(246, 244, 238)
        States.WarningDistanceText.Text = string.format("距离  %.1f m", States.WarningLastDistance or 0)
        local maxDist = math.max(1, tonumber(States.WarningDistance) or 50)
        local ratio = math.clamp((States.WarningLastDistance or maxDist) / maxDist, 0, 1)
        local armed = States.WarningCurrentPlayer.Character and States.WarningCurrentPlayer.Character:FindFirstChildOfClass("Tool") ~= nil
        if armed then States.WarningStatusText.Text = "持枪"
        elseif ratio <= 0.28 then States.WarningStatusText.Text = "高度警戒"
        elseif ratio <= 0.60 then States.WarningStatusText.Text = "接近中"
        else States.WarningStatusText.Text = "范围内" end
        States.WarningStatusText.TextColor3 = police and Color3.fromRGB(120, 180, 255) or Color3.fromRGB(229, 191, 80)
        if States.WarningStroke then
            States.WarningStroke.Color = police and (States.WarningCurrentPlayer.TeamColor and States.WarningCurrentPlayer.TeamColor.Color or Color3.fromRGB(120, 180, 255)) or Color3.fromRGB(229, 191, 80)
        end
        if States.WarningIndicator then
            States.WarningIndicator.BackgroundColor3 = police and (States.WarningCurrentPlayer.TeamColor and States.WarningCurrentPlayer.TeamColor.Color or Color3.fromRGB(120, 180, 255)) or Color3.fromRGB(235, 192, 74)
        end
        SetWarningCardScale()
    elseif not hasTarget and States.WarningTitle then
        States.WarningTitle.Text = "雷达待机"
        States.WarningTitle.TextColor3 = Color3.fromRGB(243, 242, 237)
        if States.WarningStroke then States.WarningStroke.Color = Color3.fromRGB(226, 188, 76) end
        if States.WarningIndicator then States.WarningIndicator.BackgroundColor3 = Color3.fromRGB(232, 190, 73) end
    end

    if States.WarningDots then
        local distance = math.max(0, tonumber(States.WarningLastDistance) or States.WarningDistance)
        local maxDist = math.max(1, tonumber(States.WarningDistance) or 50)
        local closeness = math.clamp(1 - distance / maxDist, 0, 1)
        local speed = hasTarget and (2.0 + closeness * 5.0) or 1.2
        RadarAnim.dotPhase = RadarAnim.dotPhase + dt * speed
        for i, dot in ipairs(States.WarningDots) do
            local offset = RadarAnim.dotPhase - (i - 1) * 0.72
            local wave = (math.sin(offset * math.pi) + 1) * 0.5
            local size = 4 + wave * 2.5
            dot.Size = UDim2.new(0, size, 0, size)
            dot.BackgroundTransparency = 0.68 - wave * 0.62
            dot.Position = UDim2.new(0, 5 + (i - 1) * 11, 0.5, -wave * 4)
        end
    end

    if States.WarningStroke then
        RadarAnim.pulsePhase = RadarAnim.pulsePhase + dt * (hasTarget and 2.5 or 1.0)
        local pulse = (math.sin(RadarAnim.pulsePhase * math.pi) + 1) * 0.5
        States.WarningStroke.Transparency = 0.08 + pulse * 0.06
        States.WarningStroke.Thickness = hasTarget and (1.5 + pulse * 0.8) or 1.5
    end

    if States.WarningIndicator then
        local indicatorPulse = hasTarget and (1 + math.sin(RadarAnim.pulsePhase * math.pi * 2) * 0.15) or 1
        States.WarningIndicator.Size = UDim2.new(0, 11 * indicatorPulse, 0, 11 * indicatorPulse)
    end
end)

task.spawn(function()
    while true do
        task.wait(RADAR_UPDATE_INTERVAL)
        if not States.ProximityWarning then
            if States.WarningVisible then HideWarningIslandCompletely() end
        else
            EnsureWarningUI()
            if not States.WarningVisible then
                States.WarningVisible = true
                RadarAnim.currentY = RADAR_HIDE_Y; RadarAnim.targetY = RADAR_SHOW_Y
                RadarAnim.currentWidth = RADAR_CARD_WIDTH_IDLE; RadarAnim.currentHeight = RADAR_CARD_HEIGHT_IDLE
                RadarAnim.currentCorner = RADAR_CARD_CORNER_IDLE
                RadarAnim.targetWidth = RADAR_CARD_WIDTH_IDLE; RadarAnim.targetHeight = RADAR_CARD_HEIGHT_IDLE
                RadarAnim.targetCorner = RADAR_CARD_CORNER_IDLE
            end
            local player, distance, isArmed = GetNearestPlayer()
            if player and distance then
                UpdateWarningData(player, distance, isArmed)
                PlayWarningBeep(distance)
            else
                States.WarningCurrentPlayer = nil; States.WarningCurrentTargetId = nil
                States.WarningLastDistance = math.huge; States.WarningNextBeep = os.clock() + 0.18
            end
        end
    end
end)

            if States.OriginalPromptDurations[prompt] == nil then States.OriginalPromptDurations[prompt] = prompt.HoldDuration end
            pcall(function() prompt.HoldDuration = 0 end)
        elseif States.OriginalPromptDurations[prompt] ~= nil then
            local old = States.OriginalPromptDurations[prompt]
            if prompt.Parent then pcall(function() prompt.HoldDuration = old end) end
            States.OriginalPromptDurations[prompt] = nil
        end
    end
    for _, obj in ipairs(workspace:GetDescendants()) do if obj:IsA("ProximityPrompt") then Apply(obj) end end
    if not v then States.OriginalPromptDurations = {} end
end })

local WaterUpdateAt = 0
RunService.Heartbeat:Connect(function(dt)
    local root, isVeh = GetControlRoot()
    local seat, model = GetVehicleContext()
    local now = os.clock()
    if seat and root and model and not States.CarFlyEnabled and not States.FreezeEnabled and not States.VehicleSpinFling then
        local maxSpeed = math.max(1, tonumber(States.VehicleConfig.MaxSpeed) or 120)
        local accel = math.max(0.01, tonumber(States.VehicleConfig.Acceleration) or 0.30)
        local brake = math.max(0.01, tonumber(States.VehicleConfig.BrakePower) or 30)
        local vel = root.AssemblyLinearVelocity
        local horizontal = Vector3.new(vel.X, 0, vel.Z)
        local speed = horizontal.Magnitude
        local throttle = tonumber(seat.ThrottleFloat) or 0
        local forward = Vector3.new(seat.CFrame.LookVector.X, 0, seat.CFrame.LookVector.Z)
        if forward.Magnitude > 0.001 then forward = forward.Unit else forward = Vector3.zero end
        if throttle > 0.05 and forward.Magnitude > 0 then
            local target = forward * maxSpeed * math.clamp(throttle, 0, 1)
            local blend = math.clamp(accel * dt * 6, 0, 1)
            horizontal = horizontal:Lerp(target, blend)
        elseif throttle < -0.05 and forward.Magnitude > 0 then
            local target = -forward * maxSpeed * math.clamp(-throttle, 0, 1)
            local blend = math.clamp(accel * dt * 6, 0, 1)
            horizontal = horizontal:Lerp(target, blend)
        elseif speed > 0.05 then
            local decay = math.exp(-brake * dt * 4)
            horizontal *= decay
            if horizontal.Magnitude < 0.35 then horizontal = Vector3.zero end
        end
        if horizontal.Magnitude > maxSpeed then horizontal = horizontal.Unit * maxSpeed end
        pcall(function() root.AssemblyLinearVelocity = Vector3.new(horizontal.X, vel.Y, horizontal.Z) end)
    end
    if not root then return end
    if States.FreezeEnabled and States.FreezePosition then
        pcall(function() root.CFrame = States.FreezePosition; root.AssemblyLinearVelocity = Vector3.zero; root.AssemblyAngularVelocity = Vector3.zero end)
    end
    if States.CarFlyEnabled and isVeh then
        local cam = Camera
        if cam then
            local currentDir = root.CFrame.LookVector
            local targetDir = cam.CFrame.LookVector
            local blended = currentDir:Lerp(targetDir, math.clamp(0.15 * (dt * 60), 0, 1))
            local newDir = blended.Magnitude > 0.001 and blended.Unit or currentDir
            local newPos = root.Position + newDir * States.CarFlySpeed * dt
            pcall(function() root.CFrame = CFrame.new(newPos, newPos + newDir); root.AssemblyLinearVelocity = Vector3.zero; root.AssemblyAngularVelocity = Vector3.zero end)
        end
    end
    if States.VehicleSpinFling and isVeh and not States.FreezeEnabled then
        local step = math.max(0, tonumber(States.FlingSpeed) or 8) * dt
        local _, yaw, _ = root.CFrame:ToOrientation()
        local pos = root.Position
        pcall(function() root.CFrame = CFrame.new(pos) * CFrame.Angles(0, yaw + step, 0); root.AssemblyLinearVelocity = Vector3.zero; root.AssemblyAngularVelocity = Vector3.zero end)
    elseif not States.VehicleSpinFling and isVeh then
        pcall(function() root.AssemblyAngularVelocity = Vector3.zero end)
    end
    if States.AutoCruise and States.TargetWaypoint and not States.FreezeEnabled and not States.VehicleSpinFling then
        local targetCF = CFrame.new(States.TargetWaypoint)
        pcall(function() root.CFrame = root.CFrame:Lerp(targetCF, math.clamp(0.05 * (dt * 60), 0, 1)); root.AssemblyLinearVelocity = Vector3.zero end)
    end
    if States.DriveOnWater and isVeh and now >= WaterUpdateAt and not States.FreezeEnabled then
        WaterUpdateAt = now + 0.08
        local ray = Ray.new(root.Position, Vector3.new(0, -15, 0))
        local hit, pos = workspace:FindPartOnWithIgnoreList(ray, model and {model} or {})
        if pos and ((hit and (hit.Name:lower():find("water") or hit.Material == Enum.Material.Water)) or pos.Y < 3) then
            if not States.WaterPlatform then
                local p = Instance.new("Part")
                p.Name = "NexbyteWaterPlatform"
                p.Size = Vector3.new(120, 1, 120)
                p.Anchored = true
                p.Transparency = 1
                p.CanCollide = true
                p.Parent = workspace
                States.WaterPlatform = p
            end
            States.WaterPlatform.Position = Vector3.new(root.Position.X, pos.Y + 2.5, root.Position.Z)
        end
    end
end)

-- ESP 绘制引擎
local espUpdateThrottle = 0
local ESP_INTERVAL = 1 / 25
local function SetupESP(player)
    if not DrawingAvailable or States.ESPDrawings[player] then return end
    local ok, drawings = pcall(function()
        local box = Drawing.new("Square")
        local nameText = Drawing.new("Text")
        local healthText = Drawing.new("Text")
        local teamText = Drawing.new("Text")
        local tracer = Drawing.new("Line")
        box.Visible = false; box.Thickness = 1
        nameText.Visible = false; nameText.Outline = true; nameText.Size = 12; nameText.Center = true
        healthText.Visible = false; healthText.Outline = true; healthText.Size = 11; healthText.Center = true
        teamText.Visible = false; teamText.Outline = true; teamText.Size = 11; teamText.Center = true
        tracer.Visible = false; tracer.Thickness = 1
        return { Box = box, Name = nameText, Health = healthText, Team = teamText, Tracer = tracer }
    end)
    if not ok or type(drawings) ~= "table" then return end
    States.ESPDrawings[player] = drawings
end
local function HideESP(drawings)
    if not drawings then return end
    for _, drawing in pairs(drawings) do pcall(function() drawing.Visible = false end) end
end
local function RemoveESP(player)
    local drawings = States.ESPDrawings[player]
    if not drawings then return end
    HideESP(drawings)
    for _, drawing in pairs(drawings) do pcall(function() drawing:Remove() end) end
    States.ESPDrawings[player] = nil
end
for _, player in ipairs(Players:GetPlayers()) do if player ~= LocalPlayer then SetupESP(player) end end
Players.PlayerAdded:Connect(function(player) if player ~= LocalPlayer then SetupESP(player) end end)
Players.PlayerRemoving:Connect(function(player)
    RemoveESP(player)
    local beam = States.BeamObjects[player]
    if beam then pcall(function() beam:Destroy() end); States.BeamObjects[player] = nil end
    States.WarnedPlayers[player.UserId] = nil
    if States.WarningCurrentPlayer == player then States.WarningCurrentPlayer = nil; StopWarningIsland() end
end)

RunService.RenderStepped:Connect(function(dt)
    espUpdateThrottle = espUpdateThrottle + dt
    if espUpdateThrottle < ESP_INTERVAL then return end
    espUpdateThrottle = 0
    if not DrawingAvailable or not States.ESP.Enabled then
        for _, drawings in pairs(States.ESPDrawings) do HideESP(drawings) end
        return
    end
    local cam = Camera
    if not cam then return end
    local viewportBottom = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y)
    for player, drawings in pairs(States.ESPDrawings) do
        local box = drawings.Box; local nameText = drawings.Name; local healthText = drawings.Health
        local teamText = drawings.Team; local tracer = drawings.Tracer
        if not player or not player.Parent or player == LocalPlayer then HideESP(drawings); continue end
        local char = player.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not hrp or not hum or hum.Health <= 0 then HideESP(drawings); continue end
        local screenPos, onScreen = cam:WorldToViewportPoint(hrp.Position)
        if not onScreen or screenPos.Z <= 0 then HideESP(drawings); continue end
        local headPos = cam:WorldToViewportPoint(hrp.Position + Vector3.new(0, 3, 0))
        local feetPos = cam:WorldToViewportPoint(hrp.Position + Vector3.new(0, -3.5, 0))
        local height = math.max(8, math.abs(feetPos.Y - headPos.Y))
        local width = math.max(4, height / 2)
        local color = player.TeamColor and player.TeamColor.Color or Color3.new(1, 1, 1)
        if States.ESP.Boxes then
            box.Visible = true; box.Size = Vector2.new(width, height)
            box.Position = Vector2.new(screenPos.X - width / 2, screenPos.Y - height / 2)
            box.Color = color
        else box.Visible = false end
        if States.ESP.Names then
            nameText.Visible = true; nameText.Text = player.DisplayName or player.Name
            nameText.Position = Vector2.new(screenPos.X, screenPos.Y - height / 2 - 16)
            nameText.Color = Color3.new(1, 1, 1)
        else nameText.Visible = false end
        if States.ESP.Health then
            healthText.Visible = true; healthText.Text = string.format("HP: %d / %d", math.floor(hum.Health), math.floor(hum.MaxHealth))
            healthText.Position = Vector2.new(screenPos.X, screenPos.Y + height / 2 + 2)
            healthText.Color = Color3.new(1, 1, 1)
        else healthText.Visible = false end
        if States.ESP.Team then
            teamText.Visible = true; teamText.Text = "[" .. ((player.Team and player.Team.Name) or "中立") .. "]"
            teamText.Position = Vector2.new(screenPos.X, screenPos.Y + height / 2 + (States.ESP.Health and 16 or 2))
            teamText.Color = color
        else teamText.Visible = false end
        if States.ESP.Tracers then
            tracer.Visible = true; tracer.From = viewportBottom
            tracer.To = Vector2.new(screenPos.X, screenPos.Y + height / 2)
            tracer.Color = color
        else tracer.Visible = false end
    end
end)

-- ==================== 清理 ====================
local function CleanupVehicleRuntime()
    for part, original in pairs(States.OriginalFrictions) do
        if part and part.Parent then pcall(function() part.CustomPhysicalProperties = original end) end
    end
    States.OriginalFrictions = {}
    for part, original in pairs(States.OriginalCollisions) do
        if part and part.Parent then pcall(function() part.CanCollide = original end) end
    end
    States.OriginalCollisions = {}
    States.SpinAngle = 0
    if States.WaterPlatform then pcall(function() States.WaterPlatform:Destroy() end); States.WaterPlatform = nil end
    if States.NeonLightObj then pcall(function() States.NeonLightObj:Destroy() end); States.NeonLightObj = nil end
    States.NeonLightRoot = nil
    States.FreezePosition = nil
    for player in pairs(States.ESPDrawings) do RemoveESP(player) end
    VehicleCache.Seat = nil; VehicleCache.Model = nil; VehicleCache.Root = nil
end

end)
