    -- 极速且安全的纯文本匹配逻辑
    local function translateText(text)
        if not text or type(text) ~= "string" or text == "" then return text end
        
        -- 1. 优先精确匹配 (O(1) 速度)
        if Translations[text] then
            return Translations[text]
        end
        
        -- 2. 子字符串匹配 (关闭正则，防语法报错)
        for en, cn in pairs(Translations) do
            if #en > 2 and text:find(en, 1, true) then
                local b, e = text:find(en, 1, true)
                return text:sub(1, b - 1) .. cn .. text:sub(e + 1)
            end
        end
        
        return text
    end

    -- 无需修改元表的安全监听引擎
    local function startSafeTranslator()
        task.spawn(function()
            local CoreGui = game:GetService("CoreGui")
            local Players = game:GetService("Players")
            local LocalPlayer = Players.LocalPlayer or Players.PlayerAdded:Wait()
            local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")
            
            -- 使用弱引用防止内存泄漏，记录已绑定的控件
            local boundObjects = setmetatable({}, {__mode = "k"})

            local function processTextElement(obj)
                if boundObjects[obj] then return end
                if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
                    boundObjects[obj] = true
                    
                    -- 首次翻译
                    pcall(function()
                        local currentText = obj.Text
                        if currentText and currentText ~= "" then
                            local newText = translateText(currentText)
                            if newText ~= currentText then
                                obj.Text = newText
                            end
                        end
                    end)
                    
                    -- 监听后续文字变化
                    obj:GetPropertyChangedSignal("Text"):Connect(function()
                        pcall(function()
                            local txt = obj.Text
                            if txt and txt ~= "" then
                                local tTxt = translateText(txt)
                                if tTxt ~= txt then
                                    obj.Text = tTxt
                                end
                            end
                        end)
                    end)
                end
            end

            local function bindContainer(container)
                if not container then return end
                container.DescendantAdded:Connect(function(descendant)
                    task.wait(0.01)
                    processTextElement(descendant)
                end)
                for _, desc in ipairs(container:GetDescendants()) do
                    processTextElement(desc)
                end
            end

            -- 监听 CoreGui 与 PlayerGui 的 UI 生成
            pcall(bindContainer, CoreGui)
            pcall(bindContainer, PlayerGui)
        end)
    end

    -- 1. 启动完全无害的汉化引擎
    startSafeTranslator()

    -- 2. 正常运行原 Luarmor 加密脚本 (不修改任何元表，不触发反作弊)
    local success, err = pcall(function()

    end)

    if not success then
        warn("Luarmor 主脚本加载失败:", err)
    end
