    -- 纯文本安全替换函数，防正则崩溃
    local function plainReplace(str, findStr, replaceStr)
        local b, e = str:find(findStr, 1, true)
        if not b then return str end
        return str:sub(1, b - 1) .. replaceStr .. str:sub(e + 1)
    end

    local function translateText(text)
        if not text or type(text) ~= "string" or text == "" then return text end
        
        -- 精确匹配优先
        if Translations[text] then
            return Translations[text]
        end
        
        -- 模糊查找（关闭正则匹配）
        for en, cn in pairs(Translations) do
            if text:find(en, 1, true) then
                return plainReplace(text, en, cn)
            end
        end
        
        return text
    end

    local function setupTranslationEngine()
        task.spawn(function()
            local success, err = pcall(function()
                local rawMT = getrawmetatable(game)
                local oldIndex = rawMT.__newindex
                setreadonly(rawMT, false)
                
                rawMT.__newindex = newcclosure(function(t, k, v)
                    if typeof(t) == "Instance" and (t:IsA("TextLabel") or t:IsA("TextButton") or t:IsA("TextBox")) and k == "Text" then
                        pcall(function()
                            v = translateText(tostring(v))
                        end)
                    end
                    return oldIndex(t, k, v)
                end)
                
                setreadonly(rawMT, true)
            end)
            
            if not success then
                warn("元表劫持失败，降级为轮询模式:", err)
                
                local function translateGui(gui)
                    if gui:IsA("TextLabel") or gui:IsA("TextButton") or gui:IsA("TextBox") then
                        pcall(function()
                            local text = gui.Text
                            if text and text ~= "" then
                                local translatedText = translateText(text)
                                if translatedText ~= text then
                                    gui.Text = translatedText
                                end
                            end
                        end)
                    end
                end

                local function bindContainer(container)
                    if not container then return end
                    container.DescendantAdded:Connect(function(descendant)
                        task.wait(0.05)
                        translateGui(descendant)
                    end)
                    for _, desc in ipairs(container:GetDescendants()) do
                        translateGui(desc)
                    end
                end

                pcall(bindContainer, game:GetService("CoreGui"))
                local player = game:GetService("Players").LocalPlayer
                if player and player:FindFirstChild("PlayerGui") then
                    pcall(bindContainer, player.PlayerGui)
                end
            end
        end)
    end

    -- 开启汉化引擎
    setupTranslationEngine()

    end)

    if not success then
        warn("主脚本加载失败:", err)
    end
